

const { MongoClient } = require("mongodb");

const url = "mongodb://127.0.0.1:27017";
const client = new MongoClient(url);

async function run() {
    try {
        await client.connect();
        console.log("Connected to MongoDB");
        const db = client.db("CompanyDB");
        const collection = db.collection("Employees");

  
        await collection.insertMany([
            { emp_id: 1, name: "Shruti", department: "IT", salary: 50000 },
            { emp_id: 2, name: "Rahul", department: "HR", salary: 30000 },
            { emp_id: 3, name: "Priya", department: "IT", salary: 60000 },
            { emp_id: 4, name: "Amit", department: "Sales", salary: 40000 },
            { emp_id: 5, name: "Neha", department: "HR", salary: 35000 }
        ]);

        console.log("Sample Data Inserted");

   

        console.log("\nDepartment Wise Total Salary:");

        const aggregationResult = await collection.aggregate([
            {
                $group: {
                    _id: "$department",
                    totalSalary: { $sum: "$salary" },
                    averageSalary: { $avg: "$salary" },
                    employeeCount: { $sum: 1 }
                }
            }
        ]).toArray();

        console.log(aggregationResult);

       

        await collection.createIndex({ name: 1 });

        console.log("\nIndex Created on 'name' field");

        // Query using indexed field
        const employee = await collection.findOne({ name: "Shruti" });

        console.log("\nEmployee Found Using Index:");
        console.log(employee);

    } catch (err) {
        console.log(err);
    } finally {
        await client.close();
    }
}

run();
