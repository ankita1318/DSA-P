class Node:
    def __init__(self, value):
        self.data = value
        self.left = None
        self.right = None

def is_operator(c):
    return c in "+-*/"

def construct_tree(prefix):
    stack = []
 
    for ch in reversed(prefix):
        if not is_operator(ch):
            stack.append(Node(ch))
        else:
            node = Node(ch)
            node.left = stack.pop()
            node.right = stack.pop()
            stack.append(node)
    return stack[-1]

def postorder_non_recursive(root):
    if root is None:
        return
    s1 = []
    s2 = []
    s1.append(root)
    while s1:
        node = s1.pop()
        s2.append(node)
        if node.left:
            s1.append(node.left)
        if node.right:
            s1.append(node.right)

    while s2:
        print(s2.pop().data, end=' ')

def delete_tree(root):
    if root is None:
        return
    delete_tree(root.left)
    delete_tree(root.right)
    del root


prefix = "+--a*bc/def"
root = construct_tree(prefix)

print("Postorder Traversal (Non-Recursive): ")
postorder_non_recursive(root)

delete_tree(root)
print("\nTree deleted successfully.")
